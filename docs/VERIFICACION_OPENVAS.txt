.venv/bin/python - <<'PY'
import json
from pathlib import Path
from datetime import datetime
from zoneinfo import ZoneInfo
for name in ('run_latest.meta.json', 'retry_latest.meta.json'):
    p=Path('runtime/openvas/artifacts')/name
    if not p.exists():
        print(name, 'todavia no existe'); continue
    m=json.loads(p.read_text()); b=m.get('backend', {})
    print(json.dumps({'archivo':name,
      'fecha_Lima':datetime.fromisoformat(m['started_at']).astimezone(ZoneInfo('America/Lima')).isoformat(),
      'recoleccion_ok':m.get('success'), 'error_captura':m.get('error'),
      'hallazgos':m.get('filtered_findings'), 'backend_enabled':b.get('enabled'),
      'alarmas_confirmadas':b.get('sent_ok'), 'snapshots_confirmados':b.get('snapshots_sent'),
      'intentos_envio':b.get('attempted'), 'repeticiones_confirmadas':b.get('replayed_snapshots'),
      'errores_backend':b.get('backend_errors'), 'errores_validacion':b.get('validation_errors'),
      'pendientes_en_ese_momento':b.get('pending_total'), 'comparacion':b.get('queued')},
      ensure_ascii=False, indent=2))

ULTIMOS SNAPSHOTS: SOLO LECTURA, NO CREA SQLITE SI FALTA
.venv/bin/python - <<'PY'
import json,sqlite3
from pathlib import Path
from datetime import datetime
from zoneinfo import ZoneInfo
p=Path('runtime/openvas/delivery.sqlite3').resolve()
if not p.exists(): raise SystemExit('No existe la cola productiva todavia')
with sqlite3.connect(p.as_uri()+'?mode=ro',uri=True) as db:
    db.row_factory=sqlite3.Row
    for row in db.execute('SELECT snapshot_id,status,attempts,next_attempt,last_error,payload FROM snapshot_events ORDER BY id DESC LIMIT 15'):
        r=dict(row); r['alarmas']=len(json.loads(r.pop('payload')).get('alarms',[]))
        epoch=r.pop('next_attempt')
        r['elegible_desde_Lima']=datetime.fromtimestamp(epoch,ZoneInfo('America/Lima')).isoformat() if r['status']=='pending' and epoch else None
        print(json.dumps(r,ensure_ascii=False))
PY

COMO INTERPRETAR
- success=true en metadata semanal confirma recoleccion, NO todo el backend.
- sent_ok cuenta alarmas confirmadas, tambien las de repeticiones reconocidas;
  no es necesariamente el numero de filas nuevas ni de hallazgos unicos.
- unchanged=1 e intentos=0 es correcto si no cambio nada y no habia pendientes.
- acked: confirmado. pending: pendiente. attempts: fallos de envio acumulados.
- Una fila acked con attempts>0 tuvo fallos anteriores y luego se confirmo.
- next_attempt es elegibilidad, no un proceso autonomo: el cron por minuto lo revisa.
- El worker silencioso sin pendientes no actualiza retry_latest.meta.json; que
  falte ese archivo no demuestra que el cron no funciona. Correlacionar con cron.
- Los reintentos NO sobrescriben el resultado semanal. Su pending_total puede
  quedar historico; consultar SQLite para saber cuantos quedan AHORA.
- No hay descargas automaticas a la laptop. *_latest se sobrescribe; SQLite
  conserva pendientes e historial confirmado 90 dias por defecto.
- No ejecutar delivery_state.py retry/force-next-snapshot durante la revision.
